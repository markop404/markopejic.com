{{- $fontsCss := "" -}} {{- range resources.Match "fonts/*/*.woff2" -}} {{- $info := split (path.BaseName .Name) "-" -}} {{- $family := replaceRE "([a-z])([A-Z])" "${1} ${2}" (index $info 0) -}} {{- $style := index $info 1 -}} {{- $weight := index $info 2 -}} {{- $url := "" -}} {{- if hugo.IsProduction -}} {{- $url = (fingerprint .).RelPermalink -}} {{- else -}} {{- $url = .RelPermalink -}} {{- end -}} {{- $fontsCss = printf ` %s @font-face { font-family: "%s"; font-style: %s; font-weight: %s; src: url("%s") format("woff2"); }` $fontsCss $family $style $weight $url -}} {{ end -}} {{- $mainCss := (resources.Get "css/main.css").Content -}} {{- $finalCss := resources.FromString "css/_.css" (printf "%s\n%s" $mainCss $fontsCss) -}} {{- $opts := dict "minify" (hugo.IsProduction) -}} {{- $css := $finalCss | css.TailwindCSS $opts -}} {{- if hugo.IsProduction -}} {{- $css = minify $css -}} {{- end -}}
